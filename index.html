import * as THREE from 'three'; // 1. 단일 건물의 2D 외곽선(Shape) 생성 예시 (사각형 형태의 건물 밑면) const buildingShape = new THREE.Shape(); buildingShape.moveTo(0, 0); buildingShape.lineTo(0, 20); // 가로/세로 크기 buildingShape.lineTo(15, 20); buildingShape.lineTo(15, 0); buildingShape.closePath(); // 2. 건물의 높이와 입체화 설정 const extrudeSettings = { steps: 1, depth: 25, // 건물의 높이 (3D로 솟아오르는 정도) bevelEnabled: true, bevelThickness: 1, bevelSize: 1, bevelSegments: 2 }; const geometry = new THREE.ExtrudeGeometry(buildingShape, extrudeSettings); // 3. 건물 재질 (회색 빌딩 느낌) const material = new THREE.MeshStandardMaterial({ color: 0xcccccc, roughness: 0.3, metalness: 0.2 }); const buildingMesh = new THREE.Mesh(geometry, material); // 건물이 바닥에 수직으로 세워지도록 회전값 조정 필요시 적용 buildingMesh.rotation.x = -Math.PI / 2; scene.add(buildingMesh);
